    function startPhase(phaseName, minutes) {

      const trueTime = Date.now() + serverOffset;

      const ms = minutes * 60 * 1000;

      db.set({

        currentPhase: phaseName,

        endTime: trueTime + ms,

        isRunning: true,

        extraTimeRequestedBy: "" 

      });

    }


    function addExtraTime(seconds) {

      db.once('value').then(snap => {

        const state = snap.val();

        if(state && state.isRunning) {

          db.update({ 

            endTime: state.endTime + (seconds * 1000),

            extraTimeRequestedBy: "" 

          });

        }

      });

    }


    function stopTimer() {

      db.update({ isRunning: false });

    }
